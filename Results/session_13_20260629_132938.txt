```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int* data;
    int size;
    int capacity;
} DynArray;

DynArray* create_array(int initial_capacity) {
    DynArray* arr = malloc(sizeof(DynArray));

    if (arr == NULL) {
        printf("Memory allocation failed\n");
        return NULL;
    }

    arr->data = malloc(initial_capacity * sizeof(int));

    if (arr->data == NULL) {
        printf("Memory allocation failed\n");
        free(arr);
        return NULL;
    }

    arr->size = 0;
    arr->capacity = initial_capacity;

    return arr;
}

void push_back(DynArray* arr, int value) {
    if (arr->size == arr->capacity) {
        int new_capacity = arr->capacity * 2;
        int* new_data = realloc(arr->data, new_capacity * sizeof(int));

        if (new_data == NULL) {
            printf("Realloc failed\n");
            return;
        }

        arr->data = new_data;
        arr->capacity = new_capacity;

        printf("Capacity doubled to: %d\n", arr->capacity);
    }

    arr->data[arr->size] = value;
    arr->size++;
}

int get(DynArray* arr, int index) {
    if (index < 0 || index >= arr->size) {
        printf("Error: index out of bounds\n");
        return -1;
    }

    return arr->data[index];
}

void free_array(DynArray* arr) {
    if (arr != NULL) {
        free(arr->data);
        free(arr);
    }
}

int main() {
    DynArray* arr = create_array(2);

    if (arr == NULL) {
        return 1;
    }

    for (int i = 0; i < 10; i++) {
        push_back(arr, i * 10);
        printf("Added %d | Size: %d | Capacity: %d\n",
               i * 10, arr->size, arr->capacity);
    }

    printf("\nArray elements:\n");

    for (int i = 0; i < arr->size; i++) {
        printf("%d ", get(arr, i));
    }

    printf("\n");

    free_array(arr);

    return 0;
}
```
