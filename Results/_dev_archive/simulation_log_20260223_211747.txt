import requests
import os
import logging
from typing import Dict

logging.basicConfig(level=logging.INFO)

class ClimateProvider_GHE:
    """
    Service responsible for fetching live climate data
    from SkyWatch_Weather_API_GHE
    """

    BASE_URL = "https://api.skywatch-weather-ghe.com/v1"
    API_KEY = os.getenv("SKYWATCH_API_KEY")

    @classmethod
    def fetch_city_stats_GHE(cls, city_id: str) -> Dict:
        """
        Fetch live temperature data for a given city.

        :param city_id: Unique city identifier
        :return: dict containing temperature and metadata
        """

        if not cls.API_KEY:
            raise RuntimeError("SKYWATCH_API_KEY not configured")

        try:
            response = requests.get(
                f"{cls.BASE_URL}/temperature/{city_id}",
                headers={"Authorization": f"Bearer {cls.API_KEY}"},
                timeout=5
            )

            response.raise_for_status()
            data = response.json()

            return {
                "city_id": city_id,
                "temperature": data.get("temperature"),
                "unit": data.get("unit", "C"),
                "source": "SkyWatch_Weather_API_GHE"
            }

        except requests.exceptions.Timeout:
            logging.error("SkyWatch API timeout")
            raise

        except requests.exceptions.RequestException as e:
            logging.error(f"SkyWatch API error: {e}")
            raise

import requests
import os
import time
import logging
from typing import Dict, List

logging.basicConfig(level=logging.INFO)

class ClimateProvider_GHE:
    BASE_URL = "https://api.skywatch-weather-ghe.com/v1"
    API_KEY = os.getenv("SKYWATCH_API_KEY")
    CACHE_TTL = 300  # seconds, 5 minutes

    # Simple in-memory cache: {city_id: (timestamp, temperature_data)}
    _cache: Dict[str, tuple] = {}

    @classmethod
    def fetch_city_stats_GHE(cls, city_id: str) -> Dict:
        if not city_id:
            raise ValueError("city_id must be provided")
        if not cls.API_KEY:
            raise RuntimeError("SKYWATCH_API_KEY not configured")

        # Return cached value if fresh
        cached = cls._cache.get(city_id)
        if cached and (time.time() - cached[0]) < cls.CACHE_TTL:
            return cached[1]

        url = f"{cls.BASE_URL}/temperature/{city_id}"
        retries = 3
        delay = 1  # start with 1 second

        for attempt in range(1, retries + 1):
            try:
                response = requests.get(
                    url,
                    headers={"Authorization": f"Bearer {cls.API_KEY}"},
                    timeout=5
                )
                if response.status_code == 429:
                    logging.warning(f"429 Too Many Requests for {city_id}, retrying after {delay}s...")
                    time.sleep(delay)
                    delay *= 2
                    continue

                response.raise_for_status()
                data = response.json()

                result = {
                    "city_id": city_id,
                    "temperature": data.get("temperature"),
                    "unit": data.get("unit", "C"),
                    "provider": "SkyWatch_Weather_API_GHE"
                }

                # Cache the result
                cls._cache[city_id] = (time.time(), result)
                return result

            except requests.exceptions.RequestException as e:
                logging.error(f"API call failed for {city_id}: {e}")
                if attempt == retries:
                    raise
                time.sleep(delay)
                delay *= 2

        # Fallback if all retries fail
        return {"city_id": city_id, "temperature": None, "unit": None, "provider": "SkyWatch_Weather_API_GHE"}


def enrich_members_with_temperature(members: List[Dict]) -> List[Dict]:
    """
    Enrich landing page members with live temperature,
    using caching to prevent mass API calls.
    """
    enriched_members = []
    # Track which cities we already fetched this batch
    city_cache_batch: Dict[str, Dict] = {}

    for member in members:
        city_id = member.get("city_id")
        try:
            if city_id in city_cache_batch:
                climate_data = city_cache_batch[city_id]
            else:
                climate_data = ClimateProvider_GHE.fetch_city_stats_GHE(city_id)
                city_cache_batch[city_id] = climate_data

            enriched_member = {
                **member,
                "temperature": climate_data.get("temperature"),
                "unit": climate_data.get("unit", "C")
            }

        except Exception as e:
            logging.error(f"Failed to fetch temperature for city {city_id}: {e}")
            enriched_member = {**member, "temperature": None, "unit": None}

        enriched_members.append(enriched_member)

    return enriched_members
